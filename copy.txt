/**
 * Rebuilds an order and adds it to the cache.
 * This method uses the complete transaction log entries from the context to reconstruct the order.
 *
 * @param context The recovery context containing the order information
 */
public void rebuildOrder(RecoveryContext context) {
    String orderId = context.getOrderId();
    RecallTicket ticket = context.getTicket();
    
    LOGGER.info("Rebuilding order {} with state {} during recovery", 
        orderId, ticket.getCurrentState());
    
    // Add ticket to cache
    recallTicketCache.put(ticket.getId(), ticket);
    
    // Get all transaction log entries for this order
    List<TransactionLogEntry> logEntries = context.getAllLogEntries();
    if (logEntries == null || logEntries.isEmpty()) {
        LOGGER.warn("No transaction log entries found for order {}, using minimal reconstruction", orderId);
        // Fall back to minimal reconstruction if no log entries are available
        reconstructOrderMinimal(ticket);
        return;
    }
    
    // Create or get corresponding order using full transaction logs
    Order order = reconstructOrderFromTransactionLogs(ticket, logEntries);
    
    if (order != null) {
        orderCache.put(order.getOrderID(), order);
        LOGGER.info("Successfully rebuilt order {} with state {}", 
            orderId, order.getCurrentState());
    } else {
        LOGGER.error("Failed to rebuild order {} during recovery", orderId);
    }
}

/**
 * Minimal order reconstruction using only the recall ticket.
 * This is a fallback method when no transaction logs are available.
 *
 * @param ticket The recall ticket
 */
private void reconstructOrderMinimal(RecallTicket ticket) {
    try {
        Order order = Order.createFromRecallTicket(ticket);
        if (order != null) {
            // Set order state based on ticket state
            OrderState orderState = mapTicketStateToOrderState(ticket.getCurrentState());
            order.setCurrentState(orderState);
            
            // Add to cache
            orderCache.put(order.getOrderID(), order);
            LOGGER.info("Minimal reconstruction completed for order {} with state {}", 
                ticket.getId(), orderState);
            
            // Create execution report for PendingFill state
            if (OrderState.PendingFill.equals(orderState)) {
                ExecutionReport report = ExecutionReport.generateDummyExecution(order);
                report.setLastQty(ticket.getFillQty());
                report.setCumQty(ticket.getFillQty());
                report.setLastPrice(ticket.getFillPrice());
                report.setAvgPrice(ticket.getFillPrice());
                report.setLeavesQty(ticket.getRecallQty() - ticket.getFillQty());
                order.setFillRequest(report);
                LOGGER.info("Created dummy ExecutionReport for PendingFill order {}", ticket.getId());
            }
        } else {
            LOGGER.error("Failed minimal reconstruction for order {}", ticket.getId());
        }
    } catch (Exception e) {
        LOGGER.error("Error during minimal reconstruction of order {}: {}", 
            ticket.getId(), e.getMessage(), e);
    }
}

/**
 * Reconstructs an order from transaction logs.
 * This follows a step-by-step approach to build a complete order object:
 * 1. Create base order from recall ticket
 * 2. Process all transaction log entries in chronological order
 * 3. Apply state changes and populate order fields based on entries
 *
 * @param ticket The recall ticket
 * @param logEntries The complete list of transaction log entries for this order
 * @return The reconstructed order, or null if reconstruction failed
 */
private Order reconstructOrderFromTransactionLogs(RecallTicket ticket, List<TransactionLogEntry> logEntries) {
    String orderId = ticket.getId();
    
    try {
        // Step 1: Create base order from recall ticket
        Order order = Order.createFromRecallTicket(ticket);
        if (order == null) {
            LOGGER.error("Failed to create base order from ticket: {}", orderId);
            return null;
        }
        
        // Set initial state from ticket
        OrderState initialState = mapTicketStateToOrderState(ticket.getCurrentState());
        order.setCurrentState(initialState);
        
        // Create a copy of the entries to avoid concurrent modification
        List<TransactionLogEntry> entriesCopy = new ArrayList<>(logEntries);
        
        // Sort entries by timestamp (oldest first)
        entriesCopy.sort(Comparator.comparing(TransactionLogEntry::getTimestamp));
        
        LOGGER.info("Processing {} transaction log entries for order {}", 
            entriesCopy.size(), orderId);
        
        // Step 2: Process entries to update order state and fields
        for (TransactionLogEntry entry : entriesCopy) {
            processTransactionLogEntry(order, entry);
        }
        
        // Final state validation and adjustment
        validateAndAdjustFinalState(order, ticket, entriesCopy);
        
        return order;
        
    } catch (Exception e) {
        LOGGER.error("Error reconstructing order {}: {}", orderId, e.getMessage(), e);
        return null;
    }
}

/**
 * Processes a single transaction log entry to update an order.
 * This method enriches the order based on the entry's state and content.
 *
 * @param order The order to update
 * @param entry The transaction log entry to process
 */
private void processTransactionLogEntry(Order order, TransactionLogEntry entry) {
    String state = entry.getState();
    String source = entry.getSource();
    
    LOGGER.debug("Processing entry for order {}: {} from {}", 
        order.getOrderID(), state, source);
    
    // Handle entries based on source and state
    if ("RECALL/TO/OMS".equals(source)) {
        processOmsRequestEntry(order, entry);
    } else if ("OMS/TO/RECALL".equals(source)) {
        processOmsResponseEntry(order, entry);
    }
}

/**
 * Processes an OMS request entry to update an order.
 * This handles entries from RECALL/TO/OMS topic.
 *
 * @param order The order to update
 * @param entry The transaction log entry to process
 */
private void processOmsRequestEntry(Order order, TransactionLogEntry entry) {
    String state = entry.getState();
    
    // Update order state
    OrderState orderState = OrderState.valueOf(state);
    order.setCurrentState(orderState);
    
    // For PendingFill state, create and attach ExecutionReport
    if (OrderState.PendingFill.equals(orderState)) {
        if (order.getFillRequest() == null) {
            ExecutionReport report = createExecutionReportFromEntry(entry);
            order.setFillRequest(report);
            LOGGER.debug("Created ExecutionReport for PendingFill order {}", order.getOrderID());
        }
    }
    
    // For PendingReplace/PendingCancel states, create and attach AmendRequest
    if (OrderState.PendingReplace.equals(orderState) || OrderState.PendingCancel.equals(orderState)) {
        if (order.getAmendRequest() == null) {
            AmendRequest amendRequest = createAmendRequestFromEntry(entry);
            order.setAmendRequest(amendRequest);
            LOGGER.debug("Created AmendRequest for {} order {}", 
                state, order.getOrderID());
        }
    }
}

/**
 * Processes an OMS response entry to update an order.
 * This handles entries from OMS/TO/RECALL topic.
 *
 * @param order The order to update
 * @param entry The transaction log entry to process
 */
private void processOmsResponseEntry(Order order, TransactionLogEntry entry) {
    String state = entry.getState();
    Character execType = entry.getExecType();
    
    // Update order state based on OMS response
    OrderState orderState = OrderState.valueOf(state);
    order.setCurrentState(orderState);
    
    // For fill-related states, update the execution report
    if (OrderState.PartiallyFilled.equals(orderState) || OrderState.Filled.equals(orderState)) {
        ExecutionReport report = order.getFillRequest();
        if (report == null) {
            report = createExecutionReportFromEntry(entry);
            order.setFillRequest(report);
        } else {
            updateExecutionReportFromEntry(report, entry);
        }
        LOGGER.debug("Updated ExecutionReport for {} order {}", 
            state, order.getOrderID());
    }
}

/**
 * Creates a new ExecutionReport from a transaction log entry.
 *
 * @param entry The transaction log entry
 * @return The created ExecutionReport
 */
private ExecutionReport createExecutionReportFromEntry(TransactionLogEntry entry) {
    ExecutionReport report = new ExecutionReport();
    
    // Set fields from entry
    report.setClOrdID(entry.getOrderId());
    report.setOrigClOrdID(entry.getOrderId());
    report.setOrderID(entry.getOrderId());
    report.setExecId(entry.getExecutionId());
    report.setExecType(entry.getExecType());
    report.setLastQty(entry.getFillQty());
    report.setCumQty(entry.getFillQty());
    report.setLeavesQty(entry.getRecallQty() - entry.getFillQty());
    report.setLastPrice(entry.getFillPrice());
    report.setAvgPrice(entry.getFillPrice());
    report.setOrderState(entry.getState());
    
    // If entry is from OMS response, extract ExecutionReport from message
    if ("OMS/TO/RECALL".equals(entry.getSource())) {
        ExecutionReport sourceReport = entry.getMessageAs(ExecutionReport.class);
        if (sourceReport != null) {
            // Copy additional fields from source report
            report.setSenderCompID(sourceReport.getSenderCompID());
            report.setTargetCompID(sourceReport.getTargetCompID());
            report.setSide(sourceReport.getSide());
            report.setSymbol(sourceReport.getSymbol());
            report.setCurrency(sourceReport.getCurrency());
            report.setTransactTime(sourceReport.getTransactTime());
            report.setSendingTime(sourceReport.getSendingTime());
            report.setTradeDate(sourceReport.getTradeDate());
            report.setHandlInst(sourceReport.getHandlInst());
            report.setIdSource(sourceReport.getIdSource());
            report.setSecurityID(sourceReport.getSecurityID());
        }
    }
    
    return report;
}

/**
 * Updates an existing ExecutionReport with data from a transaction log entry.
 *
 * @param report The ExecutionReport to update
 * @param entry The transaction log entry
 */
private void updateExecutionReportFromEntry(ExecutionReport report, TransactionLogEntry entry) {
    // Update fields from entry
    report.setExecId(entry.getExecutionId());
    report.setExecType(entry.getExecType());
    report.setLastQty(entry.getFillQty());
    report.setCumQty(entry.getFillQty());
    report.setLeavesQty(entry.getRecallQty() - entry.getFillQty());
    report.setLastPrice(entry.getFillPrice());
    report.setAvgPrice(entry.getFillPrice());
    report.setOrderState(entry.getState());
    
    // If entry is from OMS response, extract ExecutionReport from message
    if ("OMS/TO/RECALL".equals(entry.getSource())) {
        ExecutionReport sourceReport = entry.getMessageAs(ExecutionReport.class);
        if (sourceReport != null) {
            // Copy additional fields from source report
            report.setTransactTime(sourceReport.getTransactTime());
            report.setSendingTime(sourceReport.getSendingTime());
        }
    }
}

/**
 * Creates a new AmendRequest from a transaction log entry.
 *
 * @param entry The transaction log entry
 * @return The created AmendRequest
 */
private AmendRequest createAmendRequestFromEntry(TransactionLogEntry entry) {
    String orderId = entry.getOrderId();
    String newId = UUID.randomUUID().toString();
    
    // Create AmendRequest with data from entry
    AmendRequest amendRequest = new AmendRequest(
        entry.getRecallQty(),
        entry.getFillPrice(),
        newId,
        orderId
    );
    
    return amendRequest;
}

/**
 * Validates and potentially adjusts the final state of an order.
 * This handles special cases and ensures consistency.
 *
 * @param order The order to validate
 * @param context The recovery context
 */
private void validateAndAdjustFinalState(Order order, RecoveryContext context) {
    OrderState currentState = order.getCurrentState();
    String historyState = context.getHistoryState();
    OrderState historyOrderState = mapTicketStateToOrderState(historyState);
    
    // If states don't match, log warning and use history state
    if (!currentState.equals(historyOrderState)) {
        LOGGER.warn("State mismatch for order {}: calculated={}, history={} - using history state",
            order.getOrderID(), currentState, historyOrderState);
        order.setCurrentState(historyOrderState);
    }
    
    // Special validation for PendingFill orders
    if (OrderState.PendingFill.equals(order.getCurrentState()) && order.getFillRequest() == null) {
        LOGGER.warn("PendingFill order {} has no ExecutionReport - creating default", order.getOrderID());
        order.setFillRequest(ExecutionReport.generateDummyExecution(order));
    }
    
    // Special validation for PendingCancel/PendingReplace orders
    if ((OrderState.PendingCancel.equals(order.getCurrentState()) || 
         OrderState.PendingReplace.equals(order.getCurrentState())) && 
        order.getAmendRequest() == null) {
        
        LOGGER.warn("{} order {} has no AmendRequest - creating default", 
            order.getCurrentState(), order.getOrderID());
        
        String newId = UUID.randomUUID().toString();
        long recallQty = context.getTicket().getRecallQty();
        double fillPrice = context.getTicket().getFillPrice();
        
        order.setAmendRequest(new AmendRequest(recallQty, fillPrice, newId, order.getOrderID()));
    }
}

/**
 * Maps a ticket state string to the corresponding OrderState enum value.
 *
 * @param ticketState The ticket state string
 * @return The corresponding OrderState enum value
 */
private OrderState mapTicketStateToOrderState(String ticketState) {
    if (ticketState == null || ticketState.trim().isEmpty()) {
        LOGGER.error("Null or empty ticket state provided, returning New state");
        return OrderState.New;
    }
    try {
        if ("Created".equals(ticketState)) {
            return OrderState.New;
        }
        return OrderState.valueOf(ticketState);
    } catch (Exception e) {
        LOGGER.error("Error mapping ticket state '{}', defaulting to New: {}", 
            ticketState, e.getMessage(), e);
        return OrderState.New;
    }
}
